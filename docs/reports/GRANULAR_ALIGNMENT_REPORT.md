def filter_significant_variants(variants_df):
    """过滤临床显著变异"""
    return variants_df[
        (variants_df['Freq(%)'] > 5) &  # 频率 > 5%
        (
            (variants_df['Clinvar'].isin(['Pathogenic', 'Likely pathogenic'])) |
            (variants_df['有药物关联'] == True) |
            (variants_df['ExonFunc'] == 'exonic')

    ]
